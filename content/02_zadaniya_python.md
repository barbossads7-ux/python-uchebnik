### 7.2
```python
def average_grade(**students):
    result = {}
    for name, grades in students.items():
        if len(grades) == 0:
            result[name] = 0
        else:
            result[name] = round(sum(grades) / len(grades), 2)
    return result

print(average_grade(anna=[5, 4, 1, 3], boris=[5, 5]))
# {'anna': 3.25, 'boris': 5.0}
```

### 7.3
```python
def digit_sum(n):
    if n < 10:                       # базовый случай: одна цифра
        return n
    return n % 10 + digit_sum(n // 10)

def reverse(s):
    if len(s) <= 1:                  # базовый случай
        return s
    return reverse(s[1:]) + s[0]     # разворот хвоста + первая буква в конец

print(digit_sum(472))     # 13
print(reverse("привет"))  # тевирп
```
